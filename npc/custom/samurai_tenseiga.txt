// Masaru, the blacksmith's son, from the Twin Flames Saga (doc/custom_jobs/twin_flames.md).
// His ancestor forged Muramasa at the First Forge with Yami whispering over the anvil. In
// Chapter VII he reads that ancestor's ledger and finds the way back there; the forging of
// Tenseiga-no-Hayate itself is Chapter X, at the First Forge (Masaru#forge, tf_ch07_10.txt).
prontera,161,179,4	script	Masaru#tenseiga	4_M_KAGE_OLD,{
	mes "[Masaru]";
	if (Class != JOB_SAMURAI) {
		mes "My ancestor's hammer birthed a cursed blade. I carry his guilt, and his craft.";
		mes "Come back when you walk the way of the Samurai.";
		close;
	}
	// Samurai who forged Tenseiga before the saga existed keep its flame.
	if (SamuraiTenseigaForged && !getskilllv("SAM_MORTAL_FLAME")) {
		mes "Tenseiga burns with mortal will, not a god's flame. Let that flame burn in your own hands as well.";
		skill "SAM_MORTAL_FLAME", 1, SKILL_PERM;
		specialeffect2 EF_KO_IZAYOI;
		next;
		mes "[Masaru]";
		mes "You have learned ^FF0000Mortal Flame^000000. Tenseiga's own techniques are yours to master now.";
		close;
	}
	if (TF_STEP >= 120 || getequipid(EQI_HAND_R) == 900003) {
		mes "Tenseiga. No god's gift, no curse's chain.";
		mes "Whatever my family broke, you and that blade have mended.";
		close;
	}
	if (TF_STEP < 71) {
		mes "Kusanagi and Muramasa were never opposites. They are the two halves of one goddess's divided flame.";
		mes "But that is a long story, and you have only begun yours.";
		close;
	}
	if (TF_BOND & 64) {
		mes "The First Forge lies on the second level of the Magma Dungeon. When the time comes, I will be there.";
		close;
	}
	if (countitem(900021) < 5) {
		mes "So you are the one who carries Muramasa. My family made that blade. I have spent my life wishing we had not.";
		next;
		mes "[Masaru]";
		mes "My ancestor kept a ledger of every blade he forged. When he went mad, he tore it apart and scattered the pages into the fire of Thor's volcano.";
		mes "The Kasa and Salamanders there swallowed them. Bring me ^0000FFfive pages^000000 of the smith's ledger.";
		close;
	}
	delitem 900021, 5;
	mes "His hand. I would know it anywhere.";
	next;
	mes "[Masaru]";
	mes "'The First Forge, beneath the mountain of fire in the north. Two flames from one tear: the light to Kusanagi, the dark to Muramasa.'";
	next;
	mes "[Masaru]";
	mes "Two flames from one tear... Then Kusanagi was born at the same forge. They are halves of one fire.";
	mes "And what one forge split, another forging could join.";
	next;
	mes "[Masaru]";
	mes "Take this ember. My family has kept it burning since the First Forge went cold. It is the only fire that remembers how.";
	if (!checkweight(900016, 1)) {
		next;
		mes "[Masaru]";
		mes "Your pack is full. Make room for it.";
		close;
	}
	getitem 900016, 1;
	TF_BOND |= 64;
	callfunc "F_TF_Moved";
	completequest 900162;
	callfunc "F_TF_Reward", 6000000, 4000000, 150000;
	mes "^777777You received the ^0000FFEmber of the First Smith^000000.^000000";
	callfunc "F_TF_Bonded", 96, "Tetsuya will want to hear of it, back in Amatsu.";
	close;

OnInit:
	questinfo QTYPE_QUEST, QMARK_YELLOW, "Class == JOB_SAMURAI && TF_STEP == 71 && !(TF_BOND & 64)";
	end;
}
