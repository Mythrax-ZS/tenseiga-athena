// The Clash of Brothers (Chapter VI, if the Samurai sides with the goddess): a duel of
// stances, best of five. Tetsuya gives away each strike; the right guard answers it.
// TF_DUEL: 1 the Samurai won, 2 Tetsuya did.
function	script	F_TF_Duel	{
	setarray .@tell$[0],
		"His tail coils low, curling behind his heel.",
		"He drops his shoulder and grins, the way he does before a joke.",
		"His weight rocks back. His sword point dips toward your belt.";
	setarray .@strike$[0],
		"The tail sweeps for your ankles!",
		"A feint low, and then the real cut comes down from above!",
		"He lunges, a straight thrust!";
	.@won = 0;
	for (.@round = 1; .@round <= 5 && .@won < 3 && .@round - 1 - .@won < 3; .@round++) {
		.@t = rand(3);
		mes "[Clash of Brothers: " + .@round + " of 5]";
		mes "^777777" + .@tell$[.@t] + "^000000";
		next;
		.@pick = select("Leap:Guard high:Sidestep") - 1;
		mes "^777777" + .@strike$[.@t] + "^000000";
		if (.@pick == .@t) {
			.@won++;
			mes "^0000FFYou answer it cleanly. Tetsuya stumbles back, laughing.^000000";
			specialeffect2 EF_HOLYHIT;
		} else {
			mes "^FF0000It lands. Not hard. Just hard enough to make a point.^000000";
			emotion ET_KIK;
		}
		next;
	}
	TF_DUEL = (.@won >= 3) ? 1 : 2;
	mes "[Tetsuya]";
	if (TF_DUEL == 1) {
		mes "Ha! All right, all right. You read me like a scroll.";
		mes "If you can read me, you can read her. Promise me you'll try.";
		emotion ET_BEST;
	} else {
		mes "And that's the trouble with trusting something shiny. You stop watching the tail.";
		mes "I'll still watch your back. Even if you're watching hers.";
		emotion ET_SMILE;
	}
	return;
}
