//===== Description: =========================================
//= The Samurai's quest line (doc/custom_jobs/twin_flames.md): the arenas where the saga's
//= bosses are called, the checks on the blades in hand, and the saga's field monsters.
//=
//= Character variables:
//=   TF_STEP   progress through the ten chapters (see the doc for each value)
//=   TF_BOND   companions whose trials are done, as bits:
//=             1 Kiyoshi, 2 Renga, 4 Ayame, 8 Taro & Jiro, 16 Hanae, 32 the Herald, 64 Masaru
//=   TF_MERCY  times mercy was shown (the captured raider, General Mibu); colours the ending
//=   TF_TRUST  1 if the goddess was trusted over Tetsuya in Chapter VI
//============================================================

// The saga's arenas. Each boss is called by a player on its quest step (F_TF_Arena) and
// leaves again after 15 minutes if it still stands. Quest kill credit (db/custom/samurai/
// quest_db.yml) goes to the killer and the party members on the map.
-	script	TF_Arenas	-1,{
	end;

OnInit:
	// arena: quest, monster, blade needed (0 none, 1 Muramasa, 2 Twin Flames, 3 Tenseiga), where it appears
	setarray .quest[1],  900122, 900131, 900132, 900133, 900134, 900141, 900142, 900151, 900152, 900161, 900171, 900172, 900192, 900192, 900192;
	setarray .mob[1],    31903,  31904,  31905,  31906,  31907,  31908,  31909,  31910,  31911,  31912,  31913,  31914,  31915,  31916,  31917;
	setarray .need[1],   1,      1,      0,      0,      0,      1,      0,      0,      0,      2,      0,      2,      3,      3,      3;
	setarray .map$[1],   "ama_dun02", "ama_fild01", "ice_dun02", "prt_monk", "ama_dun02", "ama_dun03", "ama_dun03", "ice_dun03", "ama_in02", "thor_v02", "pay_fild08", "thor_v03", "valkyrie", "valkyrie", "valkyrie";
	setarray .x[1],      47,     200,    150,    225,    42,     120,    110,    149,    204,    90,     42,     49,     48,     48,     48;
	setarray .y[1],      53,     294,    158,    143,    52,     33,     33,     48,     165,    202,    82,     253,    67,     67,     67;
	initnpctimer;
	end;

// Bosses that stood 15 minutes leave (the next player may call them again).
OnTimer60000:
	for (.@a = 1; .@a <= 15; .@a++) {
		if (.gid[.@a] && (!unitexists(.gid[.@a]) || gettimetick(2) - .since[.@a] > 900)) {
			killmonster .map$[.@a], "TF_Arenas::OnDead" + .@a;
			.gid[.@a] = 0;
		}
	}
	initnpctimer;
	end;

OnDead1:  .gid[1] = 0;  end;
OnDead2:  .gid[2] = 0;  end;
OnDead3:  .gid[3] = 0;  end;
OnDead4:  .gid[4] = 0;  end;
OnDead5:  .gid[5] = 0;  end;
OnDead6:  .gid[6] = 0;  end;
OnDead7:  .gid[7] = 0;  end;
OnDead8:  .gid[8] = 0;  end;
OnDead9:  .gid[9] = 0;  end;
OnDead10: .gid[10] = 0; end;
OnDead11: .gid[11] = 0; end;
OnDead12:
	.gid[12] = 0;
	// The false ending: the god of ashes falls, and the goddess blesses it.
	mapannounce "thor_v03", "Kagutsuchi-no-Yami has fallen. A pale golden light pours into the Abyssal Pit...", bc_map, "0xFFD27F";
	end;

// The Weaver: radiance, then eclipse, then both. Each falls into the next.
OnDead13:
	.gid[13] = 0;
	mapannounce "valkyrie", "Amaterasu-no-Mikazuki: Light alone cannot hold you? Then be swallowed by my shadow.", bc_map, "0xC8A0FF";
	monster .map$[14], .x[14], .y[14], "Eclipsed Amaterasu", .mob[14], 1, "TF_Arenas::OnDead14";
	.gid[14] = $@mobid[0];
	.since[14] = gettimetick(2);
	end;
OnDead14:
	.gid[14] = 0;
	mapannounce "valkyrie", "Amaterasu-no-Mikazuki: Moon and sun, mother and jailer. I am both, and you are nothing!", bc_map, "0xFFD27F";
	monster .map$[15], .x[15], .y[15], "Amaterasu-no-Mikazuki", .mob[15], 1, "TF_Arenas::OnDead15";
	.gid[15] = $@mobid[0];
	.since[15] = gettimetick(2);
	end;
OnDead15:
	.gid[15] = 0;
	mapannounce "valkyrie", "Amaterasu-no-Mikazuki: Perhaps mortals need no gods... perhaps that is freedom.", bc_map, "0xFFFFFF";
	end;
}

// Calls arena getarg(0)'s boss for the attached player, if their quest is on that step and
// they hold the blade it needs. Returns 1 when the boss appears.
function	script	F_TF_Arena	{
	.@a = getarg(0);
	.@quest = getvariableofnpc(.quest[.@a], "TF_Arenas");
	if (Class != JOB_SAMURAI || isbegin_quest(.@quest) != 1 || checkquest(.@quest, HUNTING) == 2)
		return 0;
	// The Weaver's arena is busy while any of her three forms stands.
	for (.@i = .@a; .@i <= (.@a == 13 ? 15 : .@a); .@i++) {
		.@gid = getvariableofnpc(.gid[.@i], "TF_Arenas");
		if (.@gid && unitexists(.@gid)) {
			mes "^777777The battle here has already begun. Wait for it to end.^000000";
			return 0;
		}
	}
	if (!callfunc("F_TF_Need", getvariableofnpc(.need[.@a], "TF_Arenas")))
		return 0;
	if (select("Draw your blade:Not yet") != 1)
		return 0;
	.@map$ = getvariableofnpc(.map$[.@a], "TF_Arenas");
	monster .@map$, getvariableofnpc(.x[.@a], "TF_Arenas"), getvariableofnpc(.y[.@a], "TF_Arenas"), "--ja--", getvariableofnpc(.mob[.@a], "TF_Arenas"), 1, "TF_Arenas::OnDead" + .@a;
	set getvariableofnpc(.gid[.@a], "TF_Arenas"), $@mobid[0];
	set getvariableofnpc(.since[.@a], "TF_Arenas"), gettimetick(2);
	specialeffect2 EF_KO_IZAYOI;
	return 1;
}

// Whether the attached player holds the blade a step needs (0 none, 1 Muramasa,
// 2 Kusanagi and Muramasa together, 3 Tenseiga); says what is missing.
function	script	F_TF_Need	{
	.@r = getequipid(EQI_HAND_R);
	.@l = getequipid(EQI_HAND_L);
	switch (getarg(0)) {
	case 1:
		if (.@r == 900002 || .@l == 900002)
			return 1;
		mes "^777777Muramasa is not in your hand. Whatever waits here answers only to that blade.^000000";
		return 0;
	case 2:
		if (callfunc("F_TF_Twin"))
			return 1;
		mes "^777777Only the Twin Flames can face this: Kusanagi in one hand, Muramasa in the other.^000000";
		return 0;
	case 3:
		if (.@r == 900003)
			return 1;
		mes "^777777Only a blade that answers to no god can reach her. Hold Tenseiga-no-Hayate.^000000";
		return 0;
	}
	return 1;
}

// 1 if Kusanagi and Muramasa are dual wielded (either hand).
function	script	F_TF_Twin	{
	.@r = getequipid(EQI_HAND_R);
	.@l = getequipid(EQI_HAND_L);
	return (.@r == 900001 && .@l == 900002) || (.@r == 900002 && .@l == 900001);
}

// How many of the eight companions' relics the attached player carries.
function	script	F_TF_Relics	{
	setarray .@relic[0], 900010, 900011, 900012, 900013, 900014, 900015, 900016, 900017;
	for (.@i = 0; .@i < getarraysize(.@relic); .@i++)
		if (countitem(.@relic[.@i]))
			.@n++;
	return .@n;
}

// A chapter's reward: base and job experience, and zeny.
function	script	F_TF_Reward	{
	getexp getarg(0), getarg(1);
	Zeny += getarg(2);
	return;
}

// The saga moved on: its quest markers and minimap marks follow.
function	script	F_TF_Moved	{
	questinfo_refresh;
	doevent "TF_Marks::OnMark";
	return;
}

// After a companion's trial: once every trial of the chapter (arg 0, TF_BOND bits) is done,
// where to report (arg 1); until then, Tetsuya knows who still waits.
function	script	F_TF_Bonded	{
	if ((TF_BOND & getarg(0)) == getarg(0))
		mes "^777777" + getarg(1) + "^000000";
	else
		mes "^777777Others still wait on you. Tetsuya will know who.^000000";
	return;
}

// Minimap marks (viewpoint) on the saga's next step, for the map the player is on: set as a
// map loads (the loadevent mapflag on every map the saga visits) and after every step
// (F_TF_Moved, queued until the dialog ends).
-	script	TF_Marks	-1,{
	end;

OnInit:
	setarray .@maps$, "prontera", "amatsu", "ama_in01", "ama_in02", "ama_fild01", "ama_dun02", "ama_dun03", "ice_dun02", "ice_dun03", "prt_monk", "thor_v02", "thor_v03", "pay_fild08", "mag_dun02", "valkyrie";
	for (.@i = 0; .@i < getarraysize(.@maps$); .@i++)
		setmapflag .@maps$[.@i], mf_loadevent;
	end;

OnPCLoadMapEvent:
OnMark:
	if (Class != JOB_SAMURAI)
		end;
	for (.@i = 1; .@i <= 8; .@i++)
		viewpoint 2, 1, 1, .@i, 0xFFFFFF;
	@tf_mark = 0;
	switch (TF_STEP) {
	case 0:  callsub L_Mark, "prontera", 158, 179; break;
	case 1:  callsub L_Mark, "amatsu", 192, 94; break;
	case 2:
		callsub L_Mark, "ama_fild01", 174, 331;	// the shrine's door, up the stone path
		callsub L_Mark, "ama_in01", 175, 176;
		break;
	case 3:
		if (checkquest(900102, HUNTING) == 2)
			callsub L_Mark, "amatsu", 192, 94;
		break;
	case 10: callsub L_Mark, "amatsu", 185, 160; break;
	case 11:
		if (checkquest(900111, HUNTING) == 2)
			callsub L_Mark, "amatsu", 185, 160;
		break;
	case 12: callsub L_Mark, "amatsu", 187, 158; break;
	case 21:
		if (isbegin_quest(900122) == 1 && checkquest(900122, HUNTING) != 2) {
			callsub L_Mark, "ama_dun02", 45, 50;
			break;
		}
		// the riddle, or Kurogane is dead: Tetsuya
	case 20:
	case 30:
	case 60:
	case 90:
		callsub L_Mark, "amatsu", 192, 163;
		break;
	case 31: callsub L_Mark, "ama_fild01", 200, 291; break;
	case 40:
		if ((TF_BOND & 7) == 7) {
			callsub L_Mark, "amatsu", 192, 163;
			break;
		}
		if (!(TF_BOND & 1)) {
			callsub L_Mark, "amatsu", 210, 150;
			if (isbegin_quest(900132) == 1 && checkquest(900132, HUNTING) != 2)
				callsub L_Mark, "ice_dun02", 150, 155;
		}
		if (!(TF_BOND & 2)) {
			callsub L_Mark, "amatsu", 200, 140;
			if (isbegin_quest(900133) == 1 && checkquest(900133, HUNTING) != 2)
				callsub L_Mark, "prt_monk", 225, 140;
		}
		if (!(TF_BOND & 4)) {
			callsub L_Mark, "amatsu", 215, 140;
			if (isbegin_quest(900134) == 1 && checkquest(900134, HUNTING) != 2)
				callsub L_Mark, "ama_dun02", 40, 50;
		}
		break;
	case 51: callsub L_Mark, "ama_dun03", 120, 30; break;
	case 52: callsub L_Mark, "ama_dun03", 110, 30; break;
	case 61:
		if ((TF_BOND & 24) == 24) {
			callsub L_Mark, "amatsu", 192, 163;
			break;
		}
		if (!(TF_BOND & 8)) {
			callsub L_Mark, "amatsu", 225, 150;
			if (isbegin_quest(900151) == 1 && checkquest(900151, HUNTING) != 2)
				callsub L_Mark, "ice_dun03", 149, 45;
		}
		if (!(TF_BOND & 16)) {
			callsub L_Mark, "amatsu", 85, 235;	// the palace gate
			callsub L_Mark, "ama_in02", 196, 172;
			if (TF_HANAE && isbegin_quest(900152) == 1 && checkquest(900152, HUNTING) != 2)
				callsub L_Mark, "ama_in02", 204, 168;
		}
		break;
	case 71:
		if ((TF_BOND & 96) == 96) {
			callsub L_Mark, "amatsu", 192, 163;
			break;
		}
		if (!(TF_BOND & 32))
			callsub L_Mark, "thor_v02", 90, 199;
		if (!(TF_BOND & 64))
			callsub L_Mark, "prontera", 161, 179;
		break;
	case 81:  callsub L_Mark, "pay_fild08", 40, 80; break;
	case 82:  callsub L_Mark, "thor_v03", 49, 250; break;
	case 100: callsub L_Mark, "mag_dun02", 49, 40; break;
	case 110:
		callsub L_Mark, "mag_dun02", 53, 44;
		callsub L_Mark, "valkyrie", 48, 64;
		break;
	}
	end;

// A mark at (map, x, y), if the player is on that map.
L_Mark:
	if (getarg(0) != strcharinfo(3))
		return;
	@tf_mark++;
	viewpoint 1, getarg(1), getarg(2), @tf_mark, 0xFFAA00;
	return;
}
