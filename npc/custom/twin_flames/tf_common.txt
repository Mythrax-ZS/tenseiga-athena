//= Character variables:

//=   TF_MERCY  times mercy was shown (the captured raider, General Mibu, the villagers of the
//=             Crimson Fields); colours the ending
//=   TF_TRUST  1 if the goddess was trusted over Tetsuya in Chapter VI
//=   TF_DUEL   the Clash of Brothers that follows (1 won, 2 lost)

// The saga's field arenas: the companions' trials (Chapters IV, VI), the Herald (VII) and the
// Contract Binder (VIII); the other bosses live in the memorials (tf_md_*.txt). Each boss is
// called by a player on its quest step (F_TF_Arena), with the companion whose trial it is at
// their side (seen by that player), and leaves after 15 minutes if it still stands. Boss and
// companion speak as it weakens; the Glacial Yokai's ritual falters twice, and only the
// Samurai can talk the twin exorcists round (F_TF_Mediate). Quest kill credit (db/custom/
// samurai/quest_db.yml) goes to the killer and the party members on the map.
-	script	TF_Arenas	-1,{
	end;

OnInit:
	// arena: quest, monster, blade needed (0 none, 2 Twin Flames), where, who comes along
	setarray .quest[1], 900132, 900133, 900134, 900151, 900161, 900171;
	setarray .mob[1],   31905,  31906,  31907,  31910,  31912,  31913;
	setarray .need[1],  0,      0,      0,      0,      2,      0;
	setarray .map$[1],  "ice_dun02", "prt_monk", "ama_dun02", "ice_dun03", "thor_v02", "pay_fild08";
	setarray .x[1],     150,    225,    40,     149,    90,     42;
	setarray .y[1],     158,    143,    47,     51,     202,    82;
	setarray .ally$[1], "Kiyoshi Takeda#arena", "Brother Renga#arena", "Ayame#arena", "Taro#arena", "", "Kohaku#tf";
	// what the boss says at half health and near the end, and what the companion answers
	setarray .half$[1],
		"Usurper General Mibu: Takeda's army was mine the moment he trusted the wrong god!",
		"Fallen Monk Genkai: Renga prays. I listened. That is the only difference between us.",
		"Kage: Your kunoichi's clan screamed so beautifully. Shall I tell you how?",
		"",
		"Herald of Ashes: Feel them pull? Two gods, one pair of arms. Something will tear.",
		"Contract Binder: The fox is property. Property does not choose.";
	setarray .low$[1],
		"Usurper General Mibu: Mercy... Takeda, you were always the merciful one...",
		"Fallen Monk Genkai: Brother... is that you? I cannot see...",
		"Kage: The Hidden Leaf... was supposed to stay buried...",
		"",
		"Herald of Ashes: The master... will be so very hungry...",
		"Contract Binder: The contract... is... burning...";
	setarray .ahalf$[1],
		"Kiyoshi Takeda: Steady! He fights like a man who has never lost. Make him learn.",
		"Brother Renga: Genkai, hear me! There is still a way home!",
		"Ayame: Don't listen to him. Cut.",
		"",
		"",
		"Kohaku: Burn it! Burn every word of it!";
	setarray .alow$[1],
		"Kiyoshi Takeda: Hold. Let him kneel. I want to see it.",
		"Brother Renga: Go gently, brother. Go gently.",
		"Ayame: For my mother. For all of them.",
		"",
		"",
		"Kohaku: I am nobody's. Nobody's!";
	initnpctimer;
	end;

OnTimer1000:
	for (.@a = 1; .@a <= 6; .@a++) {
		if (!.gid[.@a])
			continue;
		if (!unitexists(.gid[.@a]) || gettimetick(2) - .since[.@a] > 900) {
			killmonster .map$[.@a], "TF_Arenas::OnDead" + .@a;
			callsub L_Over, .@a;
			continue;
		}
		getunitdata .gid[.@a], .@d;
		.@pct = .@d[UMOB_HP] * 100 / .@d[UMOB_MAXHP];
		if (.@pct <= 50 && !(.said[.@a] & 1)) {
			.said[.@a] |= 1;
			callsub L_Lines, .@a, .half$[.@a], .ahalf$[.@a];
			// The Herald: both blades pull at once, and the Samurai pays for it.
			if (.@a == 5 && isloggedin(.caller[.@a])) {
				sc_start SC_CURSE, 5000, 0, 10000, SCSTART_NOAVOID, .caller[.@a];
				percentheal -20, 0, .caller_cid[.@a];
				mapannounce .map$[.@a], "Kusanagi blazes white and Muramasa red, and both fight for the same arms.", bc_map, "0xFF8844";
			}
		}
		if (.@pct <= 15 && !(.said[.@a] & 2)) {
			.said[.@a] |= 2;
			callsub L_Lines, .@a, .low$[.@a], .alow$[.@a];
		}
		// The twins' ritual falters at 60% and at 30%: the ice closes over the yokai.
		if (.@a == 4 && !.falter) {
			.@due = (.@pct <= 60 && !(.said[4] & 4)) ? 4 : ((.@pct <= 30 && !(.said[4] & 8)) ? 8 : 0);
			if (.@due) {
				.said[4] |= .@due;
				.falter = 1;
				setunitdata .gid[4], UMOB_DMGIMMUNE, 1;
				npctalk "Taro: Melt it, you fool! FIRE!", "Taro#arena";
				npctalk "Jiro: Drown it, you idiot! WATER!", "Jiro#arena";
				mapannounce .map$[4], "The twins argue, and their ritual falters: ice closes over the yokai. Talk them round!", bc_map, "0x88CCFF";
			}
		}
	}
	initnpctimer;
	end;

L_Lines:
	if (getarg(1) != "")
		unittalk .gid[getarg(0)], getarg(1);
	if (getarg(2) != "")
		npctalk getarg(2), .ally$[getarg(0)];
	return;

// The arena is free again: its companion leaves the caller's sight.
L_Over:
	.@a = getarg(0);
	.gid[.@a] = 0;
	.said[.@a] = 0;
	if (.@a == 4)
		.falter = 0;
	if (.@a != 5 && .@a != 6 && .caller_cid[.@a]) {
		cloakonnpc .ally$[.@a], .caller_cid[.@a];
		if (.@a == 4)
			cloakonnpc "Jiro#arena", .caller_cid[.@a];
	}
	return;

OnDead1: callsub L_Over, 1; end;
OnDead2: callsub L_Over, 2; end;
OnDead3: callsub L_Over, 3; end;
OnDead4: callsub L_Over, 4; end;
OnDead5: callsub L_Over, 5; end;
OnDead6: callsub L_Over, 6; end;
}

// Calls arena getarg(0)'s boss for the attached player, if their quest is on that step and
// they hold the blade it needs. Returns 1 when the boss appears.
function	script	F_TF_Arena	{
	.@a = getarg(0);
	.@quest = getvariableofnpc(.quest[.@a], "TF_Arenas");
	if (Class != JOB_SAMURAI || isbegin_quest(.@quest) != 1 || checkquest(.@quest, HUNTING) == 2)
		return 0;
	.@gid = getvariableofnpc(.gid[.@a], "TF_Arenas");
	if (.@gid && unitexists(.@gid)) {
		mes "^777777The battle here has already begun. Wait for it to end.^000000";
		return 0;
	}
	if (!callfunc("F_TF_Need", getvariableofnpc(.need[.@a], "TF_Arenas")))
		return 0;
	if (select("Draw your blade:Not yet") != 1)
		return 0;
	.@map$ = getvariableofnpc(.map$[.@a], "TF_Arenas");
	monster .@map$, getvariableofnpc(.x[.@a], "TF_Arenas"), getvariableofnpc(.y[.@a], "TF_Arenas"), "--ja--", getvariableofnpc(.mob[.@a], "TF_Arenas"), 1, "TF_Arenas::OnDead" + .@a;
	set getvariableofnpc(.gid[.@a], "TF_Arenas"), $@mobid[0];
	set getvariableofnpc(.since[.@a], "TF_Arenas"), gettimetick(2);
	set getvariableofnpc(.caller[.@a], "TF_Arenas"), getcharid(3);
	set getvariableofnpc(.caller_cid[.@a], "TF_Arenas"), getcharid(0);
	.@ally$ = getvariableofnpc(.ally$[.@a], "TF_Arenas");
	if (.@ally$ != "")
		cloakoffnpcself .@ally$;
	if (.@a == 4)
		cloakoffnpcself "Jiro#arena";
	specialeffect2 EF_FIREPILLARBOMB;
	return 1;
}

// The twin exorcists at the Frozen Seal (getarg(0): "Taro" or "Jiro"). When their ritual
// falters, only the Samurai can stop them choosing between fire and water.
function	script	F_TF_Mediate	{
	.@me$ = getarg(0);
	.@other$ = (.@me$ == "Taro") ? "Jiro" : "Taro";
	if (!getvariableofnpc(.falter, "TF_Arenas")) {
		mes "[" + .@me$ + "]";
		mes (.@me$ == "Taro") ? "Fire to break the ice! Watch!" : "Water to bind it! Watch!";
		close;
	}
	mes "[" + .@me$ + "]";
	mes (.@me$ == "Taro") ? "Tell him! Fire breaks ice! Everybody knows that!" : "Tell him! Water binds what is inside! Everybody except him knows that!";
	next;
	mes "[" + .@other$ + "]";
	mes "Don't listen to " + .@me$ + "! Tell us which of us is right!";
	next;
	if (select("Fire breaks the ice. Taro is right:Water binds it. Jiro is right:Both of you, together: fire, then water") != 3) {
		mes "[" + .@other$ + "]";
		mes "See?! Even the samurai takes sides! This is why we never get anything done!";
		mes "^777777The ice over the yokai grows thicker. They are not listening yet.^000000";
		close;
	}
	mes "[Taro]";
	mes "...Together?";
	mes "[Jiro]";
	mes "...Fire to break it, water to bind it. That could work. Don't tell him I said so.";
	close2;
	set getvariableofnpc(.falter, "TF_Arenas"), 0;
	setunitdata getvariableofnpc(.gid[4], "TF_Arenas"), UMOB_DMGIMMUNE, 0;
	npctalk "Taro: Fire!", "Taro#arena";
	npctalk "Jiro: Water!", "Jiro#arena";
	specialeffect EF_FIREPILLARBOMB, AREA, "Taro#arena";
	specialeffect EF_STORMGUST, AREA, "Jiro#arena";
	mapannounce "ice_dun03", "Fire and water together: the ice over the yokai shatters!", bc_map, "0x88CCFF";
	end;
}

// Whether the attached player holds the blade a step needs (0 none, 1 Muramasa,
// 2 Kusanagi and Muramasa together, 3 Tenseiga); says what is missing.
function	script	F_TF_Need	{
	.@r = getequipid(EQI_HAND_R);
	.@l = getequipid(EQI_HAND_L);
	switch (getarg(0)) {
	case 1:
		if (.@r == 900002 || .@l == 900002)
			return 1;
		mes "^777777Muramasa is not in your hand. Whatever waits here answers only to that blade.^000000";
		return 0;
	case 2:
		if (callfunc("F_TF_Twin"))
			return 1;
		mes "^777777Only the Twin Flames can face this: Kusanagi in one hand, Muramasa in the other.^000000";
		return 0;
	case 3:
		if (.@r == 900003)
			return 1;
		mes "^777777Only a blade that answers to no god can reach her. Hold Tenseiga-no-Hayate.^000000";
		return 0;
	}
	return 1;
}

// 1 if Kusanagi and Muramasa are dual wielded (either hand).
function	script	F_TF_Twin	{
	.@r = getequipid(EQI_HAND_R);
	.@l = getequipid(EQI_HAND_L);
	return (.@r == 900001 && .@l == 900002) || (.@r == 900002 && .@l == 900001);
}

// How many of the eight companions' relics the attached player carries.
function	script	F_TF_Relics	{
	setarray .@relic[0], 900010, 900011, 900012, 900013, 900014, 900015, 900016, 900017;
	for (.@i = 0; .@i < getarraysize(.@relic); .@i++)
		if (countitem(.@relic[.@i]))
			.@n++;
	return .@n;
}

// The saga moved on: its quest markers and minimap marks follow.

// After a companion's trial: once every trial of the chapter (arg 0, TF_BOND bits) is done,
// where to report (arg 1); until then, Tetsuya knows who still waits.
function	script	F_TF_Bonded	{
	if ((TF_BOND & getarg(0)) == getarg(0))
		mes "^777777" + getarg(1) + "^000000";
	else
		mes "^777777Others still wait on you. Tetsuya will know who.^000000";
	return;
}

// Opens saga memorial getarg(0) (db/custom/samurai/instance_db.yml) and moves the attached
// player in at once, so close their dialog first. It is a solo instance (IM_CHAR) that lasts an
// hour, so a player who falls inside can come back to it. Another memorial still open is let go
// first, if the player agrees. Returns 1 when the player is in.
function	script	F_TF_Memorial	{
	.@md$ = getarg(0);
	.@id = instance_id(IM_CHAR);
	if (.@id > 0 && instance_live_info(ILI_NAME, .@id) != .@md$) {
		mes "^777777Another memory still holds you: " + instance_live_info(ILI_NAME, .@id) + ". Let it go?^000000";
		if (select("Let it go:Not yet") != 1)
			return 0;
		instance_destroy .@id;
		.@id = 0;
	}
	if (.@id <= 0)
		.@id = instance_create(.@md$, IM_CHAR);
	if (.@id <= 0) {
		mes "^777777The memory will not open. Try again in a moment.^000000";
		return 0;
	}
	// instance_enter looks for a party's instance unless it is told which one.
	if (instance_enter(.@md$, -1, -1, getcharid(0), .@id) != IE_OK) {
		mes "^777777The memory will not let you in.^000000";
		return 0;
	}
	return 1;
}

// The attached player says getarg(0) aloud, over their head.
function	script	F_TF_Say	{
	unittalk getcharid(3), strcharinfo(0) + ": " + getarg(0);
	return;
}

// Minimap marks (viewpoint) on the saga's next step, for the map the player is on: set as a
// map loads (the loadevent mapflag on every map the saga visits) and after every step
// (F_TF_Moved, queued until the dialog ends).
-	script	TF_Marks	-1,{
	end;

OnInit:
	setarray .@maps$, "prontera", "amatsu", "ama_in01", "ama_in02", "ama_fild01", "ama_dun02", "ama_dun03", "ice_dun02", "ice_dun03", "prt_monk", "thor_v02", "thor_v03", "pay_fild08", "mag_dun02", "tf_goddess";
	for (.@i = 0; .@i < getarraysize(.@maps$); .@i++)
		setmapflag .@maps$[.@i], mf_loadevent;
	end;

OnPCLoadMapEvent:
OnMark:
	// Memorials (instance maps, named "<slot>#<instance>") keep their own marks.
	if (Class != JOB_SAMURAI || compare(strcharinfo(3), "#"))
		end;
	for (.@i = 1; .@i <= 8; .@i++)
		viewpoint 2, 1, 1, .@i, 0xFFFFFF;
	@tf_mark = 0;
	switch (TF_STEP) {
	case 0:  callsub L_Mark, "prontera", 158, 179; break;
	case 1:  callsub L_Mark, "amatsu", 192, 94; break;
	case 2:
	case 3:
		callsub L_Mark, "ama_fild01", 174, 331;	// the shrine's door, up the stone path
		callsub L_Mark, "ama_in01", 175, 176;
		break;
	case 4:
		if (checkquest(900102, HUNTING) == 2)
			callsub L_Mark, "amatsu", 192, 94;
		break;
	case 10: callsub L_Mark, "amatsu", 185, 160; break;
	case 11:
		if (checkquest(900111, HUNTING) == 2)
			callsub L_Mark, "amatsu", 185, 160;
		break;
	case 12: callsub L_Mark, "amatsu", 187, 158; break;
	case 21:
		if (isbegin_quest(900122) == 1) {
			callsub L_Mark, "ama_dun02", 45, 50;
			break;
		}
		// the riddle, or Kurogane is dead: Tetsuya
	case 20:
	case 30:
	case 60:
	case 90:
		callsub L_Mark, "amatsu", 192, 163;
		break;
	case 31: callsub L_Mark, "ama_fild01", 200, 291; break;
	case 40:
		if ((TF_BOND & 7) == 7) {
			callsub L_Mark, "amatsu", 192, 163;
			break;
		}
		if (!(TF_BOND & 1)) {
			callsub L_Mark, "amatsu", 198, 151;
			if (isbegin_quest(900132) == 1 && checkquest(900132, HUNTING) != 2)
				callsub L_Mark, "ice_dun02", 150, 155;
		}
		if (!(TF_BOND & 2)) {
			callsub L_Mark, "amatsu", 200, 140;
			if (isbegin_quest(900133) == 1 && checkquest(900133, HUNTING) != 2)
				callsub L_Mark, "prt_monk", 225, 140;
		}
		if (!(TF_BOND & 4)) {
			callsub L_Mark, "amatsu", 215, 140;
			if (isbegin_quest(900134) == 1 && checkquest(900134, HUNTING) != 2)
				callsub L_Mark, "ama_dun02", 40, 50;
		}
		break;
	case 51:
	case 52: callsub L_Mark, "ama_dun03", 120, 30; break;
	case 61:
		if ((TF_BOND & 24) == 24) {
			callsub L_Mark, "amatsu", 192, 163;
			break;
		}
		if (!(TF_BOND & 8)) {
			callsub L_Mark, "amatsu", 225, 150;
			if (isbegin_quest(900151) == 1 && checkquest(900151, HUNTING) != 2)
				callsub L_Mark, "ice_dun03", 149, 45;
		}
		// Hanae, who takes the Samurai into the masquerade.
		if (!(TF_BOND & 16)) {
			callsub L_Mark, "amatsu", 85, 235;	// the palace gate
			callsub L_Mark, "ama_in02", 196, 172;
		}
		break;
	case 71:
		if ((TF_BOND & 96) == 96) {
			callsub L_Mark, "amatsu", 192, 163;
			break;
		}
		if (!(TF_BOND & 32))
			callsub L_Mark, "thor_v02", 90, 199;
		if (!(TF_BOND & 64))
			callsub L_Mark, "prontera", 161, 179;
		break;
	case 81:  callsub L_Mark, "pay_fild08", 40, 80; break;
	case 82:  callsub L_Mark, "thor_v03", 49, 250; break;
	case 100: callsub L_Mark, "mag_dun02", 49, 40; break;
	case 110: callsub L_Mark, "mag_dun02", 53, 44; break;
	}
	end;

// A mark at (map, x, y), if the player is on that map.
L_Mark:
	if (getarg(0) != strcharinfo(3))
		return;
	@tf_mark++;
	viewpoint 1, getarg(1), getarg(2), @tf_mark, 0xFFAA00;
	return;
}
