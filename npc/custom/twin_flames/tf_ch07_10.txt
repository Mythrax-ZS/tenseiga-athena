	mes "[Altar of Ashes]";

		mes "^777777An altar of charred timber, heaped with ash. The heat here is unbearable.^000000";
		close;
	}
	if (checkquest(900161, HUNTING) != 2) {
		mes "^777777Yami's cult has made this altar its heart. Its Herald answers when the Twin Flames draw near.^000000";
		mes "^777777(The quest also asks for thirty Cultists of Ashes, on the volcano's first level.)^000000";
		next;
		if (callfunc("F_TF_Arena", 5)) {
			mes "[Herald of Ashes]";
			mes "Both blades in one pair of hands. The master said you would come. He is so very hungry.";
		}
		close;
	}
	mes "^777777The Herald is ash. Kusanagi blazes white and Muramasa red, and both fight for your arms at once.^000000";
	next;
	specialeffect2 EF_DARKCASTING;
	mes "^777777Pain lances up your shoulders, a backlash of two gods pulling against each other.^000000";
	mes "^777777For a moment you see a woman in moonlight and a man in flame, standing back to back.^000000";
	next;
	mes "^777777Then it is gone. You are on your knees on hot stone, still holding both.^000000";
	TF_BOND |= 32;
	callfunc "F_TF_Moved";
	completequest 900161;
	callfunc "F_TF_Reward", 12000000, 9000000, 300000;
	callfunc "F_TF_Bonded", 96, "Tetsuya will want to hear of it, back in Amatsu.";
	close;

OnInit:
	questinfo QTYPE_QUEST, QMARK_YELLOW, "Class == JOB_SAMURAI && isbegin_quest(900161) == 1 && !(TF_BOND & 32)";
	end;
}

// Kohaku, a spirit fox once bound by contract to watch the Samurai for the goddess.

	mes "[Kohaku]";
	if (Class != JOB_SAMURAI || TF_STEP < 81) {
		mes "^777777A white fox watches you from the grass. Its tails flicker like candle flames.^000000";
		close;
	}
	if (TF_STEP > 81) {
		mes "My foxfire is yours. Let it light something better than a goddess's lantern.";
		close;
	}
	if (checkquest(900171, HUNTING) != 2) {
		mes "So the monkey finally sent you. I have been following you since the volcano. Since before that, if I am honest.";
		next;
		mes "[Kohaku]";
		mes "I was bound by contract to watch you, and to report everything to the goddess. Every step. Every doubt.";
		mes "I do not want to anymore. But the contract has a keeper, and the keeper will not let me go.";
		next;
		mes "[Kohaku]";
		mes "Burn it for me. The Contract Binder answers when I call its name.";
		next;
		if (callfunc("F_TF_Arena", 6)) {
			mes "[Contract Binder]";
			mes "The fox is the goddess's property. As are you, little vessel. As are you.";
		}
		close;
	}
	mes "It is gone. The contract is ash. I am... free?";
	next;
	mes "[Kohaku]";
	mes "Then I will choose what to do with my fire. Take my foxfire; it can burn what even gods cannot cut.";
	next;
	mes "[Kohaku]";
	mes "And hurry. Yami knows you have both blades. He waits in the ^0000FFAbyssal Pit^000000, the bottom of Thor's volcano. He means to take them back, and you with them.";
	if (!checkweight(900017, 1)) {
		next;
		mes "[Kohaku]";
		mes "Your pack is too full for fire. Make room.";
		close;
	}
	getitem 900017, 1;
	completequest 900171;
	setquest 900172;
	TF_STEP = 82;
	callfunc "F_TF_Moved";
	callfunc "F_TF_Reward", 10000000, 7000000, 250000;
	mes "^777777You received ^0000FFKohaku's Foxfire^000000.^000000";
	close;

OnInit:
	questinfo QTYPE_QUEST, QMARK_YELLOW, "Class == JOB_SAMURAI && TF_STEP == 81";
	end;
}

// ---------------------------------------------------------------- X. The Mortal Flame

// The First Forge, where Kusanagi and Muramasa were both born. Masaru reforges them here
// into Tenseiga-no-Hayate (900003), with every companion's relic in the fire.
mag_dun02,49,40,4	script	Masaru#forge	4_M_KAGE_OLD,{
	mes "[Masaru]";
	if (Class != JOB_SAMURAI || TF_STEP < 100) {
		mes "The First Forge. My ancestor's hammer rang here, with Yami whispering over his anvil.";
		close;
	}
	if (TF_STEP > 100) {
		mes "Tenseiga-no-Hayate. Not a relic of gods, but the first true sword of men.";
		mes "Tetsuya is waiting for you. The way to the Celestial Veil is open.";
		close;
	}
	mes "You came. And you brought them all: the steel, the curse, the light, and the people who walked beside you.";
	next;
	mes "[Masaru]";
	if (!callfunc("F_TF_Twin")) {
		mes "Hold both blades as one, Kusanagi in one hand and Muramasa in the other. The forge must take them from your hands.";
		close;
	}
	if (callfunc("F_TF_Relics") < 8 || !countitem(900018)) {
		mes "The forge needs every relic: the eight your companions gave you, and Tetsuya's Sarugami essence.";
		mes "^777777You carry " + callfunc("F_TF_Relics") + " of the eight relics" + (countitem(900018) ? " and the essence" : ", and not the essence") + ".^000000";
		close;
	}
	if (countitem("Oridecon") < 10 || countitem("Burning_Heart") < 10) {
		mes "The fire is cold without the steel and the fuel. Bring ^0000FF10 Oridecon^000000 and ^0000FF10 Burning Hearts^000000.";
		close;
	}
	for (.@i = 0; .@i < 4; .@i++) {
		if (getequipcardid(EQI_HAND_R, .@i) || getequipcardid(EQI_HAND_L, .@i)) {
			mes "Pull the cards from your blades first. The fire would only burn them.";
			close;
		}
	}
	.@refine = min(getequiprefinerycnt(EQI_HAND_R), getequiprefinerycnt(EQI_HAND_L));
	mes "Kusanagi and Muramasa will be gone for good, and every relic with them. The new blade will be refined to ^0000FF+" + .@refine + "^000000, the lower of the two.";
	if (select("Light the forge:Not yet") != 1) {
		mes "The fire will wait. So will they.";
		close;
	}
	if (!checkweight("Samurai_Tenseiga", 1)) {
		mes "Make room in your inventory and lighten your load first.";
		close;
	}
	mes "Then we begin. And not alone.";
	next;
	// Everyone whose relic goes into the fire comes to see it go (seen by this Samurai).
	setarray .@who$[0], "Kohaku#forge", "Ayame#forge", "Brother Renga#forge", "Kiyoshi Takeda#forge", "Chiyo#forge", "Taro#forge", "Jiro#forge", "Lady Hanae#forge";
	for (.@i = 0; .@i < getarraysize(.@who$); .@i++)
		cloakoffnpcself .@who$[.@i];
	mes "[Masaru]";
	mes "Kohaku's foxfire lights the First Forge, freed from the goddess's contract. My family's ember feeds it.";
	npctalk "Kohaku: Burn bright. Burn for no one but yourself.", "Kohaku#forge", bc_self;
	specialeffect EF_FIREPILLARBOMB;
	next;
	setarray .@line$[0],
		"Ayame: My clan's kunai. Clean, at last. Go.",
		"Brother Renga: Beads that turned aside one god's whisper. Let them turn aside two.",
		"Kiyoshi Takeda: My family's crest. Honour that answers to no lord. Take it into the fire.",
		"Chiyo: A talisman with only my name on it. That is the only kind I trust now.",
		"Taro: Fire to break it...",
		"Jiro: ...and water to bind it. Don't tell him I said it was his idea.",
		"Lady Hanae: The jade from my father's fan. It has seen through every lie but this one. Let it see through hers.";
	setarray .@speaker$[0], "Ayame#forge", "Brother Renga#forge", "Kiyoshi Takeda#forge", "Chiyo#forge", "Taro#forge", "Jiro#forge", "Lady Hanae#forge";
	mes "[Masaru]";
	mes "One by one, then. Each of you, give the fire what you carried.";
	close2;
	for (.@i = 0; .@i < getarraysize(.@line$); .@i++) {
		npctalk .@line$[.@i], .@speaker$[.@i], bc_self;
		specialeffect EF_FIREPILLARBOMB;
		sleep2 2200;
	}
	specialeffect EF_MAGNUMBREAK;
	mes "[Tetsuya]";
	mes "Sarugami blood broke Kusanagi's seal. Let it seal nothing ever again. Balance is not obedience. Balance is freedom.";
	specialeffect2 EF_HOLYHIT;
	next;
	mes "[Masaru]";
	mes "And the curse's fragments, and the light's essence, from your own hands. Now, Samurai. Clash them together.";
	next;
	// Re-check: everything must still be at hand when the forge takes it.
	if (!callfunc("F_TF_Twin") || callfunc("F_TF_Relics") < 8 || !countitem(900018) || countitem("Oridecon") < 10 || countitem("Burning_Heart") < 10) {
		mes "[Masaru]";
		mes "The fire died down. Come back with both blades in hand and everything the forge needs.";
		close;
	}
	delequip EQI_HAND_R;
	delequip EQI_HAND_L;
	delitem "Oridecon", 10;
	delitem "Burning_Heart", 10;
	setarray .@relic[0], 900010, 900011, 900012, 900013, 900014, 900015, 900016, 900017, 900018;
	for (.@i = 0; .@i < getarraysize(.@relic); .@i++)
		delitem .@relic[.@i], 1;
	specialeffect2 EF_KO_IZAYOI;
	mes "[" + strcharinfo(0) + "]";
	mes "If gods are but liars, then I will find a power that is neither divine nor damned.";
	mes "No god's gift, no curse's chain. This is the blade of men, and with it, I sever heaven's lies.";
	getitem2 "Samurai_Tenseiga", 1, 1, .@refine, 0, 0, 0, 0, 0;
	SamuraiTenseigaForged = 1;
	if (!getskilllv("SAM_MORTAL_FLAME"))
		skill "SAM_MORTAL_FLAME", 1, SKILL_PERM;
	completequest 900191;
	setquest 900192;
	TF_STEP = 110;
	callfunc "F_TF_Moved";
	announce strcharinfo(0) + " has forged Tenseiga-no-Hayate, the Mortal Flame.", bc_all;
	next;
	mes "[Masaru]";
	mes "Tenseiga-no-Hayate. Not a relic of gods, but the first true sword of men.";
	mes "It takes both hands. It is two blades made one.";
	mes "And its flame burns in you now: you have learned ^FF0000Mortal Flame^000000. Tenseiga's own techniques are yours to master.";
	mes "^777777Beside the anvil, Tetsuya is already looking up, as if he can see through the mountain to the sky.^000000";
	close;

OnInit:
	questinfo QTYPE_QUEST, QMARK_YELLOW, "Class == JOB_SAMURAI && TF_STEP == 100";
	end;
}
