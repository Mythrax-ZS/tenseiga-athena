prontera,158,179,4	script	Ashigaru Courier#tf	8_M_JPNSOLDIER,5,5,{
	mes "[Ashigaru Courier]";
	if (Class != JOB_SAMURAI) {
		mes "I carry word from Amatsu, for a samurai who served a lord there.";
		mes "It is not for you, traveller.";
		close;
	}
	if (TF_STEP > 0) {
		mes "Amatsu, samurai. The old servant Oharu waits by the guides at the town's heart.";
		close;
	}
	mes "Samurai! At last. I have run from Amatsu with the worst of news.";
	emotion ET_HUK;
	next;
	mes "[Ashigaru Courier]";
	mes "Kagemori is ash. The house you served, the village in the hills above Amatsu. Raiders came by night, while you were away.";
	mes "Your lord... your kin... none of them lived.";
	next;
	mes "[Ashigaru Courier]";
	mes "Only old Oharu, who kept the kitchen, hid and survived. She asked for you by name.";
	if (select("Go to Amatsu at once:I need a moment") != 1) {
		mes "Of course. Grief does not wait for anyone, but it will wait for you.";
		close;
	}
	TF_STEP = 1;
	callfunc "F_TF_Moved";
	setquest 900101;
	mes "She waits by the town guides in Amatsu. Go swiftly, and may the kami walk with you.";
	close;

// He calls out to a Samurai passing by, once.
OnTouch:
	if (Class != JOB_SAMURAI || TF_STEP || @tf_courier)
		end;
	@tf_courier = 1;
	npctalk "Ashigaru Courier: Samurai! Samurai, wait! I have word from Kagemori!", strnpcinfo(0), bc_self;
	emotion ET_HUK;
	end;

OnInit:
	questinfo QTYPE_QUEST, QMARK_YELLOW, "Class == JOB_SAMURAI && TF_STEP == 0";
	end;
}

// The shrine's "kami of justice" is Kagutsuchi-no-Yami in disguise. He shows the Samurai the
// night Kagemori burned (the memorial in tf_md_kagemori.txt), and in it gives Muramasa.
ama_in01,175,176,4	script	Hooded Kami#tf	4_M_SOULREAPER,{
	if (Class != JOB_SAMURAI || TF_STEP < 2) {
		mes "^777777A hooded figure kneels at the altar, unmoving. Is it even breathing?^000000";
		close;
	}
	if (TF_STEP > 3) {
		mes "^777777The altar is cold. The hooded figure is gone, if it was ever here.^000000";
		close;
	}
	if (TF_STEP == 3) {
		cutin "soul_van01", 2;
		mes "[Hooded Kami]";
		mes "The night is still burning. Will you go back into it?";
		next;
		if (select("Go back:Not yet") == 1) {
			// instance_enter moves the Samurai at once: the farewell comes first.
			mes "[Hooded Kami]";
			mes "Close your eyes.";
			close2;
			cutin "", 255;
			if (!callfunc("F_TF_Memorial", "Twin Flames: Kagemori Burning"))
				close;
			end;
		}
		mes "[Hooded Kami]";
		mes "It will wait. Fire is patient.";
		close3;
	}
	mes "^777777You kneel before the forgotten altar. You do not pray for revenge.^000000";
	mes "^777777You pray for strength, to protect what is left.^000000";
	close2;
	progressbar "ffffff", 3;
	specialeffect EF_DEVIL;
	cutin "soul_van01", 2;
	mes "[Hooded Kami]";
	mes "I hear you. Few still come to this place.";
	mes "I am a kami of justice, and you prayed to me truly.";
	next;
	mes "[Hooded Kami]";
	mes "You were not there when Kagemori burned. That is the wound, is it not? Not that they died. That you were away.";
	next;
	mes "[Hooded Kami]";
	mes "I can take you there. Not to change it. To walk through it, and come out holding what you need.";
	next;
	if (select("Show me:Not yet") != 1) {
		mes "[Hooded Kami]";
		mes "Then pray again, when you are ready to look.";
		close3;
	}
	// instance_enter moves the Samurai at once: the farewell comes first.
	mes "[Hooded Kami]";
	mes "Close your eyes.";
	close2;
	cutin "", 255;
	if (!callfunc("F_TF_Memorial", "Twin Flames: Kagemori Burning"))
		close;
	completequest 900101;
	setquest 900103;
	TF_STEP = 3;
	callfunc "F_TF_Moved";
	end;

OnInit:
	questinfo QTYPE_QUEST, QMARK_YELLOW, "Class == JOB_SAMURAI && (TF_STEP == 2 || TF_STEP == 3)";
	end;
}

amatsu,187,158,4	script	Captured Raider#tf	8_M_JPNSOLDIER,3,3,{

	next;
	mes "^FF0000...he burned them. Feed me. Feed me, and you will be whole again...^000000";
	next;
	switch (select("Spare him:Cut him down")) {
	case 1:
		mes "^777777You lower the blade. It fights you, heavier than it has ever been.^000000";
		mes "^777777Then the whisper falls silent, sullen.^000000";
		next;
		mes "[Captured Raider]";
		mes "...Thank you. I will not forget this. I swear it on my daughter.";
		mes "And samurai... the one with the tail, up past the elder. He has watched you since you came to town.";
		close2;
		TF_MERCY++;
		cloakoffnpcself "Raider's Daughter#tf";
		npctalk "Raider's Daughter: Father! Father!", "Raider's Daughter#tf", bc_self;
		emotion ET_CRY, getnpcid(0, "Raider's Daughter#tf");
		break;
	case 2:
		specialeffect EF_BLOODDRAIN;
		mes "^777777Muramasa moves almost before you do. The man falls without a sound.^000000";
		mes "^777777The blade feels warm, satisfied. You feel nothing at all, and that frightens you.^000000";
		close2;
		cloakoffnpcself "Raider's Daughter#tf";
		npctalk "Raider's Daughter: ...Father?", "Raider's Daughter#tf", bc_self;
		emotion ET_SURPRISE, getnpcid(0, "Watchman#tf1");
		emotion ET_SURPRISE, getnpcid(0, "Watchman#tf2");
		sleep2 2500;
		emotion ET_CRY, getnpcid(0, "Raider's Daughter#tf");
		npctalk "Somewhere just past Elder Genzo, someone whistles, low and long.", "Captured Raider#tf", bc_self;
		break;
	}
	completequest 900112;
	TF_STEP = 20;
	callfunc "F_TF_Moved";
	callfunc "F_TF_Reward", 2000000, 1500000, 50000;
	mes "^777777Chapter II, The Wandering Ronin, is complete.^000000";
	mes "^777777Someone has been watching from the rooftops all along, a few paces past Elder Genzo...^000000";
	close;

		end;

	end;

	questinfo QTYPE_QUEST, QMARK_YELLOW, "Class == JOB_SAMURAI && TF_STEP == 12";
	end;
