function	script	F_TF_Twins	{
	.@me$ = getarg(0);
	.@other$ = (.@me$ == "Taro") ? "Jiro" : "Taro";
	mes "[" + .@me$ + "]";
	if (Class != JOB_SAMURAI || TF_STEP < 61) {
		mes (.@me$ == "Taro") ? "Fire purifies. Everyone knows that." : "Water purifies. Everyone except my brother knows that.";
		close;
	}
	if (TF_BOND & 8) {
		mes "Fire and water, one sigil. Don't tell " + .@other$ + " I said it was a good idea.";
		close;
	}
	if (checkquest(900151, HUNTING) != 2) {
		mes "A yokai woke in the deepest ice cave and sealed itself in a glacier. We've been arguing about how to unseal it for a week.";
		next;
		mes "[" + .@other$ + "]";
		mes "Because he wants to melt it and I want to drown it.";
		next;
		mes "[" + .@me$ + "]";
		mes "You! Be the tiebreaker. Go to the third floor of the Ice Dungeon and break the frozen seal. Whatever comes out, we'll seal it properly.";
		close;
	}
	mes "It's done? You fought it, and we sealed it... together?";
	next;
	mes "[" + .@other$ + "]";
	mes "Fire to break the ice, water to bind what was inside. It needed both of us.";
	next;
	mes "[" + .@me$ + "]";
	mes "Take our sigils, joined. Fire and water, balanced. Don't make us say it twice.";
	if (!checkweight(900014, 1)) {
		next;
		mes "[" + .@me$ + "]";
		mes "Your pack is full. Come back lighter.";
		close;
	}
	getitem 900014, 1;
	TF_BOND |= 8;
	callfunc "F_TF_Moved";
	completequest 900151;
	callfunc "F_TF_Reward", 6000000, 4000000, 150000;
	mes "^777777You received the ^0000FFTwin Sigils of Fire and Water^000000.^000000";
	callfunc "F_TF_Bonded", 24, "Tetsuya will want to hear of it. He keeps to the rooftops by Elder Genzo.";
	close;
}
