//===== rAthena Script =======================================
//= The Twin Flames Saga: The Crimson Fields (memorial dungeon, Chapter IV)
//===== Description: =========================================
//= Through the Scorched War Banner (tf_ch04_06.txt). Kurogane's death broke the truce, and the
//= clan armies are tearing the fields apart with villagers caught between them. Cut down
//= thirty soldiers; the villagers cowering among them are not soldiers, but Muramasa does not
//= care, and every one that falls feeds what waits beneath the banner. Then the Shade of
//= Yami, Yami's own voice, the blade that will not leave the hand, and Tetsuya's tail.
//= TF_STEP 31 -> 40. The villagers spared or slain colour Chapter IV's end (TF_MERCY).
//= Instance "Twin Flames: The Crimson Fields" on tf_crimson (a copy of ama_fild01).
//=
//= Instance variables:
//=   'map$     the instance's copy of tf_crimson    'aid      the Samurai's account id
//=   'stage    0 the line, 1 the banner, 2 the Shade, 3 Yami speaks, 4 over
//=   'fallen   soldiers cut down                     'blood    villagers slain
//=   'boss     the Shade's GID while it stands       'phase    bits: 1 the pull, 2 the soldiers
//============================================================

tf_crimson	mapflag	loadevent
tf_crimson	mapflag	noteleport
tf_crimson	mapflag	nomemo
tf_crimson	mapflag	nobranch
tf_crimson	mapflag	monster_noteleport
tf_crimson	mapflag	nosave	ama_fild01,200,288

tf_crimson,1,1,0	script	#cf_ctrl	-1,{
	end;

OnPCLoadMapEvent:
	.@id = instance_id(IM_CHAR);
	if (.@id > 0) {
		if (strcharinfo(3) == instance_mapname("tf_crimson", .@id))
			doevent instance_npcname("#cf_ctrl", .@id) + "::OnEnter";
	}
	end;

OnInstanceInit:
	'map$ = instance_mapname("tf_crimson");
	disablenpc instance_npcname("Tetsuya#cf");
	for (.@i = 1; .@i <= 4; .@i++)
		cloakonnpc instance_npcname("#cf_dead" + .@i);
	// Villagers cowering in the open, among the soldiers.
	setarray .@vx[0], 130, 160, 190, 215, 241, 150, 205, 250;
	setarray .@vy[0], 150, 175, 140, 190, 160, 205, 120, 200;
	for (.@i = 0; .@i < 8; .@i++)
		monster 'map$, .@vx[.@i], .@vy[.@i], "Cowering Villager", (.@i % 2 ? 31933 : 31922), 1, instance_npcname("#cf_ctrl") + "::OnVillagerDead";
	donpcevent instance_npcname("#cf_ctrl") + "::OnWave";
	end;

OnEnter:
	'aid = getcharid(3);
	playBGM ('stage >= 2 ? "156" : "73");
	addtimer 1000, instance_npcname("#cf_ctrl") + "::OnScene";
	end;

OnScene:
	if (!'intro) {
		'intro = 1;
		setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL, true;
		mes "^777777Smoke over the rice terraces. Two armies, and between them, everyone who used to live here.^000000";
		mes "^777777Muramasa hums in your hand. It has been humming since you stepped onto the field.^000000";
		next;
		mes "^FF0000...so many. So many. Cut. Cut them all. Soldier or farmer, what does it matter? All are meat.^000000";
		close2;
		setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL, false;
		callfunc "F_TF_Say", "Quiet. I fight soldiers. Only soldiers.";
		instance_announce 0, "Cut down thirty Crimson Fields soldiers. Villagers are cowering among them: spare them.", bc_map, "0xFFAA66";
	}
	callsub L_Marks;
	end;

// Soldiers come in waves until thirty have fallen.
OnWave:
	if ('stage)
		end;
	.@ev$ = instance_npcname("#cf_ctrl") + "::OnSoldierDead";
	.@n = min(8, 30 - 'fallen) - mobcount('map$, .@ev$);
	if (.@n > 0)
		areamonster 'map$, 110, 115, 255, 215, "--ja--", 31901, .@n, .@ev$;
	end;

OnSoldierDead:
	if ('stage)
		end;
	'fallen++;
	if ('fallen < 30) {
		if ('fallen % 10 == 0)
			instance_announce 0, "Soldiers cut down: " + 'fallen + " of 30.", bc_map, "0xFFAA66";
		if (mobcount('map$, instance_npcname("#cf_ctrl") + "::OnSoldierDead") < 3)
			donpcevent instance_npcname("#cf_ctrl") + "::OnWave";
		end;
	}
	// The line breaks. In the quiet, the Samurai sees the field for what it is.
	'stage = 1;
	killmonster 'map$, instance_npcname("#cf_ctrl") + "::OnSoldierDead";
	killmonster 'map$, instance_npcname("#cf_ctrl") + "::OnVillagerDead";
	for (.@i = 1; .@i <= 4; .@i++)
		cloakoffnpc instance_npcname("#cf_dead" + .@i);
	if ('blood)
		instance_announce 0, "The field goes quiet. Among the soldiers lie " + 'blood + " who were not soldiers. Your blade is red to the hilt.", bc_map, "0xFF4444";
	else
		instance_announce 0, "The field goes quiet. The villagers you spared run for the river. Muramasa sulks in your hand.", bc_map, "0xFFAA66";
	soundeffectall "st_woman_cry.wav", 0, 'map$;
	sleep 3000;
	instance_announce 0, "Something stronger waits beneath the scorched banner to the north-east. The blade is pulling you toward it.", bc_map, "0xFFAA66";
	if (attachrid('aid))
		callsub L_Marks;
	end;

OnVillagerDead:
	'blood++;
	setarray .@line$[0],
		"...yes. Yes! There is no difference. Feel how warm it is?",
		"...more. Did you see her face? Neither did I.",
		"...good. Good. Meat is meat.";
	instance_announce 0, .@line$[rand(3)], bc_map, "0xFF2222";
	end;

// The Shade rises from the banner: Yami's hunger, fed by every villager who fell.
OnShade:
	'stage = 2;
	monster 'map$, 265, 285, "Shade of Yami", 31904, 1, instance_npcname("#cf_ctrl") + "::OnShadeDead";
	'boss = $@mobid[0];
	if ('blood) {
		getunitdata 'boss, .@d;
		setunitdata 'boss, UMOB_ATKMIN, .@d[UMOB_ATKMIN] * (10 + 'blood) / 10;
		setunitdata 'boss, UMOB_ATKMAX, .@d[UMOB_ATKMAX] * (10 + 'blood) / 10;
	}
	unittalk 'boss, "Shade of Yami: Hungry, little ronin? So am I." + ('blood ? " And you have fed me so well." : "");
	playBGMall "156", 'map$;
	donpcevent instance_npcname("#cf_watch") + "::OnStart";
	end;

OnShadeSoldierDead:
	end;

// A villager struck down during the pull feeds the Shade.
OnPull:
	'blood++;
	if ('boss && unitexists('boss)) {
		getunitdata 'boss, .@d;
		setunitdata 'boss, UMOB_HP, min(.@d[UMOB_MAXHP], .@d[UMOB_HP] + .@d[UMOB_MAXHP] / 10);
		unittalk 'boss, "Shade of Yami: Ahh... sweet. Again.";
	}
	end;

// Yami speaks (the Samurai attached): the blade will not leave the hand, and Tetsuya comes.
OnShadeDead:
	'boss = 0;
	if ('stage >= 3)
		end;
	'stage = 3;
	killmonster 'map$, instance_npcname("#cf_ctrl") + "::OnShadeSoldierDead";
	killmonster 'map$, instance_npcname("#cf_ctrl") + "::OnPull";
	if (!playerattached()) {
		if (!attachrid('aid))
			end;
	}
	setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL, true;
	cutin "se_moon02", 2;
	mes "[Kagutsuchi-no-Yami]";
	mes "Why do you weep? You did well. All are meat. All are fuel.";
	mes "Feed me, and I will make you whole again.";
	next;
	mes "[Kagutsuchi-no-Yami]";
	mes "Do you not know my voice? You prayed at my altar. ^FF0000It was I who answered your prayers.^000000";
	mes "You prayed to me, not heaven.";
	next;
	cutin "", 255;
	mes "^777777You try to throw the blade away.^000000";
	close2;
	progressbar "ff0000", 3;
	mes "^777777Your fingers will not open.^000000";
	next;
	mes "[" + strcharinfo(0) + "]";
	mes "If gods are but liars, then I will find a power that is neither divine nor damned!";
	mes "If fate has chained me, I will break it with my own hand!";
	next;
	mes "[Kagutsuchi-no-Yami]";
	mes "Break it? With what? With ^FF0000me^000000?";
	close2;
	enablenpc instance_npcname("Tetsuya#cf");
	npctalk "Tetsuya: Hey! HEY! Look at me, not at the sword!", instance_npcname("Tetsuya#cf");
	sleep2 1500;
	specialeffect2 EF_STORM_MIN;
	mes "^777777A tail snaps around your wrist and wrenches your arm down. The whisper breaks.^000000";
	next;
	mes "[Tetsuya]";
	mes "Breathe. It's a sword. A hungry, lying sword. And you are not its meal.";
	next;
	mes "[Tetsuya]";
	mes "If gods made you their pawn, then let us break their game together.";
	close2;
	setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL, false;
	'stage = 4;
	if (TF_STEP == 31) {
		if (!'blood)
			TF_MERCY++;
		completequest 900131;
		setquest 900132;
		setquest 900133;
		setquest 900134;
		TF_STEP = 40;
		callfunc "F_TF_Moved";
		callfunc "F_TF_Reward", 8000000, 6000000, 200000;
	}
	callsub L_Marks;
	end;

OnMarks:
	callsub L_Marks;
	end;
L_Marks:
	viewpoint 2, 1, 1, 1, 0xFFFFFF;
	if ('stage == 1 || 'stage == 2)
		viewpoint 1, 265, 288, 1, 0xFF0000;
	else if ('stage == 4)
		viewpoint 1, 262, 286, 1, 0x66CCFF;
	return;
}

// The Shade's phases, every second while it stands.
tf_crimson,1,2,0	script	#cf_watch	-1,{
	end;
OnStart:
	initnpctimer;
	end;
OnTimer1000:
	if (!'boss || !unitexists('boss)) {
		stopnpctimer;
		end;
	}
	getunitdata 'boss, .@d;
	.@pct = .@d[UMOB_HP] * 100 / .@d[UMOB_MAXHP];
	if (.@pct <= 66 && !('phase & 1)) {
		// The pull: Muramasa drags the Samurai's guard down, and villagers stumble into reach.
		'phase |= 1;
		unittalk 'boss, "Shade of Yami: Here. Fresh ones. Go on. Nobody is watching.";
		instance_announce 0, "Muramasa drags at your arm. Villagers stumble out of the smoke: do not strike them.", bc_map, "0xFF2222";
		sc_start SC_CONFUSION, 4000, 0, 10000, SCSTART_NOAVOID, 'aid;
		areamonster 'map$, .@d[UMOB_X] - 6, .@d[UMOB_Y] - 6, .@d[UMOB_X] + 6, .@d[UMOB_Y] + 6, "Cowering Villager", 31922, 2, instance_npcname("#cf_ctrl") + "::OnPull";
		areamonster 'map$, .@d[UMOB_X] - 6, .@d[UMOB_Y] - 6, .@d[UMOB_X] + 6, .@d[UMOB_Y] + 6, "Cowering Villager", 31933, 2, instance_npcname("#cf_ctrl") + "::OnPull";
	}
	if (.@pct <= 33 && !('phase & 2)) {
		'phase |= 2;
		unittalk 'boss, "Shade of Yami: Soldiers! Your enemy is here! Kill the ronin and feast!";
		areamonster 'map$, .@d[UMOB_X] - 7, .@d[UMOB_Y] - 7, .@d[UMOB_X] + 7, .@d[UMOB_Y] + 7, "--ja--", 31901, 4, instance_npcname("#cf_ctrl") + "::OnShadeSoldierDead";
	}
	initnpctimer;
	end;
}

// The scorched banner: the Shade waits beneath it.
tf_crimson,262,291,4	script	Scorched War Banner#cf	1_FLAG_LION,6,6,{
	mes "[Scorched War Banner]";
	if ('stage < 1) {
		mes "^777777A clan war banner, scorched and torn. The blade wants you to come back here later. Much later. When you are hungrier.^000000";
		close;
	}
	if ('stage >= 2) {
		mes "^777777The banner hangs in rags.^000000";
		close;
	}
	mes "^777777Beneath the banner, the trampled earth is moving.^000000";
	close;
OnTouch:
	if ('stage != 1)
		end;
	donpcevent instance_npcname("#cf_ctrl") + "::OnShade";
	doevent instance_npcname("#cf_ctrl") + "::OnMarks";
	end;
}

tf_crimson,180,160,2	script	#cf_dead1	4_M_DSTMANDEAD,{
	mes "^777777A farmer, still holding his hoe.^000000";
	close;
}
tf_crimson,205,182,6	duplicate(#cf_dead1)	#cf_dead2	4_VR_SWORDMAN_DEAD
tf_crimson,152,140,4	duplicate(#cf_dead1)	#cf_dead3	4_M_DSTMANDEAD

tf_crimson,259,287,6	script	Tetsuya#cf	4_EXJOB_NINJA2,{
	mes "[Tetsuya]";
	if ('stage < 4) {
		mes "Still with me? Good. Don't listen to it. Whatever it says, it's hungry, and it's lying.";
		close;
	}
	mes "Three people in Amatsu have reasons to hate the dark god as much as we do. Come on. Back to town.";
	next;
	if (select("Back to Amatsu:Not yet") == 1) {
		warp "amatsu", 192, 160;
		instance_destroy;
	}
	close;
}
